cmake_minimum_required(VERSION 3.16)
project(ModeratedLMS VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SQLite3 REQUIRED)
find_package(OpenSSL REQUIRED)
add_executable(moderated_lms src/main.cpp)
target_link_libraries(moderated_lms PRIVATE SQLite::SQLite3 OpenSSL::Crypto)
if(MSVC)
  target_compile_options(moderated_lms PRIVATE /W4)
else()
  target_compile_options(moderated_lms PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME database_initialization COMMAND moderated_lms --init-only)
